# ── 2. Ajouter les colonnes utiles ────────────────────────────
def enrich(df):
    df = df.copy()
    
    # Buts marqués / encaissés du point de vue de chaque équipe
    df["goal_for"]     = df["home_score"]
    df["goal_against"] = df["away_score"]
    
    # Résultat : W / D / L pour home_team
    df["result"] = np.where(df["home_score"] > df["away_score"], "W",
                   np.where(df["home_score"] == df["away_score"], "D", "L"))
    
    # Poids du tournoi (compétitions importantes = plus de poids)
    tournament_weights = {
        "FIFA World Cup"              : 4.0,
        "UEFA Euro"                   : 3.5,
        "Africa Cup of Nations"       : 3.5,
        "African Cup of Nations"      : 3.5,
        "Copa América"                : 3.5,
        "UEFA Nations League"         : 2.5,
        "FIFA World Cup qualification": 2.0,
        "AFC Asian Cup"               : 3.0,
        "CAF Africa Cup of Nations"   : 3.5,
    }
    df["tournament_weight"] = df["tournament"].map(tournament_weights).fillna(1.5)
    
    # Time decay : matchs récents = plus de poids
    # Référence = aujourd'hui
    ref_date = pd.Timestamp("2026-06-17")
    df["days_ago"] = (ref_date - df["date"]).dt.days
    df["time_weight"] = np.exp(-df["days_ago"] / 365)  # demi-vie = 1 an
    
    # Poids final combiné
    df["weight"] = df["tournament_weight"] * df["time_weight"]
    
    return df

df_clean = enrich(df_clean)

# ── Étape 3 : Calcul xG pondéré par équipe ────────────────────

def compute_xg(df, team):
    """
    Calcule les buts moyens marqués et encaissés
    pondérés par tournament_weight × time_weight
    """
    # Matchs où l'équipe joue à domicile
    home = df[df["home_team"] == team].copy()
    home["scored"]   = home["home_score"]
    home["conceded"] = home["away_score"]

    # Matchs où l'équipe joue à l'extérieur
    away = df[df["away_team"] == team].copy()
    away["scored"]   = away["away_score"]
    away["conceded"] = away["home_score"]

    all_matches = pd.concat([home, away], ignore_index=True)

    if len(all_matches) == 0:
        return None

    # Moyenne pondérée
    total_weight = all_matches["weight"].sum()
    xg_for     = (all_matches["scored"]   * all_matches["weight"]).sum() / total_weight
    xg_against = (all_matches["conceded"] * all_matches["weight"]).sum() / total_weight

    # Forme récente : 5 derniers matchs compétitifs
    recent = all_matches.sort_values("date").tail(5)
    recent_goals = recent["scored"].mean()
    recent_conceded = recent["conceded"].mean()

    # Win rate pondéré
    all_matches["win"] = (all_matches["scored"] > all_matches["conceded"]).astype(int)
    win_rate = (all_matches["win"] * all_matches["weight"]).sum() / total_weight

    return {
        "team"           : team,
        "matches"        : len(all_matches),
        "xg_for"         : round(xg_for, 3),
        "xg_against"     : round(xg_against, 3),
        "recent_goals"   : round(recent_goals, 3),
        "recent_conceded": round(recent_conceded, 3),
        "win_rate"       : round(win_rate, 3),
    }

# Calcul pour les deux équipes
stats_maroc    = compute_xg(df_clean, "Morocco")

from scipy.stats import poisson

# ── Étape 4 : Calcul des lambdas & modèle de Poisson ──────────

# Moyenne de buts dans le dataset global (référence ligue mondiale)
total_weight = df_clean["weight"].sum()
league_avg_home = (df_clean["home_score"] * df_clean["weight"]).sum() / total_weight
league_avg_away = (df_clean["away_score"] * df_clean["weight"]).sum() / total_weight

print(f"Moyenne mondiale buts domicile : {league_avg_home:.3f}")
print(f"Moyenne mondiale buts extérieur: {league_avg_away:.3f}")

# ── Force d'attaque & défense relative ────────────────────────
# Maroc
mar_att_strength  = stats_maroc["xg_for"]     / league_avg_home
mar_def_strength  = stats_maroc["xg_against"] / league_avg_away

# Écosse
sco_att_strength  = stats_scotland["xg_for"]     / league_avg_away
sco_def_strength  = stats_scotland["xg_against"] / league_avg_home

print(f"\n🇲🇦  Maroc   — Attaque : {mar_att_strength:.3f} | Défense : {mar_def_strength:.3f}")
print(f"🏴󠁧󠁢󠁳󠁣󠁴󠁿  Écosse  — Attaque : {sco_att_strength:.3f} | Défense : {sco_def_strength:.3f}")

# ── Lambda final (buts attendus par équipe) ────────────────────
# Maroc joue "à domicile neutre" (terrain neutre = moyenne des deux)
# Écosse joue en déplacement

# Facteur climatique : Maroc habitué chaleur, Écosse non
# Match au USA (juin) → température ~30°C, humidité élevée
climate_boost_maroc    = 1.08   # +8% Maroc (habitué chaud)
climate_penalty_ecosse = 0.93   # -7% Écosse (climat froid d'origine)

lambda_maroc   = (mar_att_strength * sco_def_strength * league_avg_home
                  * climate_boost_maroc)

lambda_scotland = (sco_att_strength * mar_def_strength * league_avg_away
                   * climate_penalty_ecosse)

print(f"\n📊 Lambda Maroc   (buts attendus) : {lambda_maroc:.3f}")
print(f"📊 Lambda Écosse  (buts attendus) : {lambda_scotland:.3f}")

# ── Probabilités Poisson ───────────────────────────────────────
MAX_GOALS = 8
prob_matrix = np.zeros((MAX_GOALS+1, MAX_GOALS+1))

for i in range(MAX_GOALS+1):
    for j in range(MAX_GOALS+1):
        prob_matrix[i][j] = poisson.pmf(i, lambda_maroc) * poisson.pmf(j, lambda_scotland)

p_maroc_win = np.sum(np.tril(prob_matrix, -1))   # i > j
p_draw      = np.sum(np.diag(prob_matrix))         # i == j
p_sco_win   = np.sum(np.triu(prob_matrix, 1))      # j > i

print(f"\n{'='*40}")
print(f"  🇲🇦  Victoire Maroc  : {p_maroc_win*100:.1f}%")
print(f"  🤝  Match nul       : {p_draw*100:.1f}%")
print(f"  🏴󠁧󠁢󠁳󠁣󠁴󠁿  Victoire Écosse : {p_sco_win*100:.1f}%")
print(f"{'='*40}")

# ── Top 5 scénarios ───────────────────────────────────────────
scenarios = []
for i in range(MAX_GOALS+1):
    for j in range(MAX_GOALS+1):
        scenarios.append((i, j, prob_matrix[i][j]))

scenarios.sort(key=lambda x: x[2], reverse=True)

print("\n🏆 TOP 5 SCÉNARIOS LES PLUS PROBABLES :")
for i, j, p in scenarios[:5]:
    label = "✅ Maroc gagne" if i > j else ("🤝 Nul" if i == j else "❌ Écosse gagne")
    print(f"  Maroc {i} – {j} Écosse : {p*100:.2f}%  {label}")
